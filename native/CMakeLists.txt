cmake_minimum_required(VERSION 3.16)
project(laplace_temporal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MUNDO_X_ROOT "${CMAKE_CURRENT_LIST_DIR}/../deps/mundo-X" CACHE PATH "Pinned original Mundo X checkout")
if(NOT EXISTS "${MUNDO_X_ROOT}/native/include/mundo_x/quantum_runtime.hpp")
  message(FATAL_ERROR "Pinned original Mundo X headers are required")
endif()
add_executable(laplace_temporal laplace_temporal.cpp)
target_include_directories(laplace_temporal PRIVATE "${MUNDO_X_ROOT}/native/include")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(laplace_temporal PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
